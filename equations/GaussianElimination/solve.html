List<double> solve()
override

Solves the Ax = b equation and returns the x vector.

This method uses the specific algorithm implemented by the concrete solver class to find the solution vector x that satisfies the equation Ax = b.

Returns a list of double values representing the solution vector x.

Throws a SystemSolverException if:

  • The matrix is singular or nearly singular (direct methods), or
  • The algorithm diverges or doesn't converge within its maximum number of steps (iterative methods).

Example:

final solver = GaussianElimination(
  matrix: RealMatrix.fromData(
    rows: 2,
    columns: 2,
    data: [[2, 1], [1, 3]],
  ),
  knownValues: [5, 7],
);

final solution = solver.solve();
// Returns: [1.6, 1.8] (approximately)

Implementation

@override
List<double> solve() {
  final n = size;
  final a = matrix.flattenData.toList(growable: false);
  final b = knownValues.toList(growable: false);
  final tolerance = precision * maxAbsValue(a);

  // Forward elimination with partial pivoting
  for (var p = 0; p < n; ++p) {
    var pivotRow = p;
    var pivotAbs = a[n * p + p].abs();
    for (var i = p + 1; i < n; ++i) {
      final value = a[n * i + p].abs();
      if (value > pivotAbs) {
        pivotAbs = value;
        pivotRow = i;
      }
    }

    if (pivotAbs <= tolerance) {
      throw const SystemSolverException(
        'The matrix is singular or nearly singular.',
      );
    }

    if (pivotRow != p) {
      for (var j = p; j < n; ++j) {
        final temp = a[n * p + j];
        a[n * p + j] = a[n * pivotRow + j];
        a[n * pivotRow + j] = temp;
      }

      final temp = b[p];
      b[p] = b[pivotRow];
      b[pivotRow] = temp;
    }

    final pivot = a[n * p + p];
    for (var i = p + 1; i < n; ++i) {
      final factor = a[n * i + p] / pivot;
      if (factor == 0) {
        continue;
      }

      b[i] -= factor * b[p];
      for (var j = p + 1; j < n; ++j) {
        a[n * i + j] -= factor * a[n * p + j];
      }
    }
  }

  final solutions = List<double>.filled(n, 0);
  for (var i = n - 1; i >= 0; --i) {
    var sum = b[i];
    for (var j = i + 1; j < n; ++j) {
      sum -= a[n * i + j] * solutions[j];
    }
    solutions[i] = sum / a[n * i + i];
  }

  return solutions;
}
